class Solution {
    int rows;
    int cols;
    int oldColor;
    int newColor;
    public int[][] floodFill(int[][] img, int sr, int sc, int color) {
        rows=img.length;
        cols=img[0].length;
        oldColor=img[sr][sc];
        newColor=color;
        if(oldColor==newColor) return img;
        dfs(img,sr,sc);
        return img;
    }
    public void dfs(int[][] img, int r, int c){
            if(r<0 || c<0 || r>=rows || c>=cols) return;
            if(img[r][c]!=oldColor) return;
            img[r][c]=newColor;
            dfs(img,r-1,c);
            dfs(img,r+1,c);
            dfs(img,r,c-1);
            dfs(img,r,c+1);
        }
}
