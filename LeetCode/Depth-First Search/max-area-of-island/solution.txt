class Solution {
    int rows;
    int cols;
    int max=0;
    int cnt=0;
    public int maxAreaOfIsland(int[][] grid) {
        rows=grid.length;
        cols=grid[0].length;
        for(int i=0;i<rows;i++){
            for(int j=0;j<cols;j++){
                if(grid[i][j]==1){
                    cnt=0;
                    dfs(grid,i,j);
                    max=Math.max(max,cnt);
                }
            }
        }
        return max;
    }
    public void dfs(int[][] grid, int r,int c){
        if(r<0 || c<0 || r>=rows || c>=cols) return;
        if(grid[r][c]==0) return;
        grid[r][c]=0;
        dfs(grid,r-1,c);
        dfs(grid,r+1,c);
        dfs(grid,r,c-1);
        dfs(grid,r,c+1);
        cnt++;
    }
}
