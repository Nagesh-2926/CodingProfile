class Solution {
    public int numSubarraysWithSum(int[] nums, int goal) {
        return need(nums,goal)-need(nums,goal-1);
    }
    private int need(int[] nums,int goal){
        if(goal<0) return 0;
        int n=nums.length;
        int sum=0;
        int l=0;
        int cnt=0;
        for(int r=0;r<n;r++){
            sum+=nums[r];
            while(sum>goal){
                sum-=nums[l++];
            }
            cnt+=r-l+1;
        }
        return cnt;
    }
}
