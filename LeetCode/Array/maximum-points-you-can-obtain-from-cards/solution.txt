class Solution {
    public int maxScore(int[] nums, int k) {
        int n=nums.length;
        int[] pre=new int[k+1];
        int[] suf=new int[k+1];
        pre[0]=0;
        suf[0]=0;
        for(int i=1;i<=k;i++){
            pre[i]=pre[i-1]+nums[i-1];
            suf[i]=suf[i-1]+nums[n-i];
        }
        int ans=0;
        for(int i=0;i<=k;i++){
            ans=Math.max(ans,pre[i]+suf[k-i]);
        }
        return ans;
    }
}
