class Solution {
    public int minSubArrayLen(int target, int[] nums) {
        int n=nums.length;
        long[] pre=new long[n+1];
        for(int i=0;i<n;i++){
            pre[i+1]=pre[i]+nums[i];
        }
        int ans=Integer.MAX_VALUE;
        for(int i=0;i<=n;i++){
            long need=pre[i]+target;
            int pos=lowerBound(pre,need);
            if(pos!=-1) ans=Math.min(ans,pos-i);
        }
        return ans==Integer.MAX_VALUE ? 0 : ans;
    }
    private int lowerBound(long[] arr,long target){
        int l=0,r=arr.length-1;
        int ans=-1;
        while(l<=r){
            int mid=l+(r-l)/2;
            if(arr[mid]>=target){
                ans=mid;
                r=mid-1;
            }else{
                l=mid+1;
            }
        }
        return ans;
    }
}
