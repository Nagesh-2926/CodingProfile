class Solution {
    public int climbStairs(int n) {
        int[] dp=new int[n+1];
        Arrays.fill(dp,-1);
        return climbing(0,n,dp);
    }
    public int climbing(int curr,int n,int[] dp){
        if(curr==n) return 1;
        if(curr>n) return 0;
        if(dp[curr]!=-1) return dp[curr];
        int res=climbing(curr+1,n,dp)+climbing(curr+2,n,dp);
        dp[curr]=res;
        return res;
    }
}
