class Solution {
    public int tribonacci(int n) {
        int[] dp=new int[n+1];
        Arrays.fill(dp,-1);
        return rec(n,dp);
    }
    public int rec(int n,int[] dp){
        if(n<=1) return n;
        if(n==2) return 1;
        if(dp[n]!=-1) return dp[n];
        int res=rec(n-2,dp)+rec(n-1,dp)+rec(n-3,dp);
        dp[n]=res;
        return res;
    }
}
