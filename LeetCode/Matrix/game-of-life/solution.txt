class Solution {
    public void gameOfLife(int[][] board) {
        int n=board.length;
        int m=board[0].length;
        int[] rowDir={-1,-1,-1,0,0,1,1,1};
        int[] colDir={-1,0,1,-1,1,-1,0,1};
        for(int row=0;row<n;row++){
            for(int col=0;col<m;col++){
                int cnt=0;
                for(int dir=0;dir<8;dir++){
                    int newRow=row+rowDir[dir];
                    int newCol=col+colDir[dir];
                    if(newRow>=0 && newRow<n && newCol>=0 && newCol<m && (board[newRow][newCol]==1 || board[newRow][newCol]==2)){
                        cnt++;
                    }
                }
                if(board[row][col]==1){
                    if(cnt<2||cnt>3) board[row][col]=2;
                }else{
                    if(cnt==3) board[row][col]=3;
                }
            }
        }
        for(int row=0;row<n;row++){
            for(int col=0;col<m;col++){
                if(board[row][col]==2) board[row][col]=0;
                else if(board[row][col]==3) board[row][col]=1;
            }
        }
    }
}
