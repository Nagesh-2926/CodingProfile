class Solution {
    public List<Integer> spiralOrder(int[][] mat) {
        List<Integer> ans=new ArrayList<>();
        int m=mat.length;
        int n=mat[0].length;
        int sr=0;
        int sc=0;
        int er=m-1;
        int ec=n-1;
        int count=0;
        while(sr<=er && sc<=ec){
            //starting row
            for(int i=sc;i<=ec;i++){
                ans.add(mat[sr][i]);
            }
            sr++;
            //ending col
            for(int j=sr;j<=er;j++){
                ans.add(mat[j][ec]);
            }
            ec--;
            // ending row
            if(sr<=er){
                for(int i=ec;i>=sc;i--){
                    ans.add(mat[er][i]);
                }
                er--;
            }
            // starting column
            if(sc<=ec){
                for(int j=er;j>=sr;j--){
                    ans.add(mat[j][sc]);
                }
                sc++;
            } 
        }
        return ans;
    }
}
