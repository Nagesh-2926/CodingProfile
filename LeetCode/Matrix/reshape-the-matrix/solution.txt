class Solution {
    public int[][] matrixReshape(int[][] mat, int r, int c) {
        if(mat.length*mat[0].length != r*c) return mat;
        int[] temp=new int[mat.length*mat[0].length];
        int idx=0;
        for(int i=0;i<mat.length;i++){
            for(int j=0;j<mat[0].length;j++){
                temp[idx++]=mat[i][j];
            }
        }
        int[][] res=new int[r][c];
        idx=0;

        for(int i=0;i<r;i++){
            for(int j=0;j<c;j++){
                res[i][j]=temp[idx++];
            }
        }
        return res;
    }
}
