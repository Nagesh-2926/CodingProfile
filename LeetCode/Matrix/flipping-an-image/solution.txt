class Solution {
    public int[][] flipAndInvertImage(int[][] mat) {
        int n=mat.length;
        for(int i=0;i<n;i++){
            int l=0;
            int r=n-1;
            while(l<=r){
                int temp=mat[i][l];
                mat[i][l]=mat[i][r]^1;
                mat[i][r]=temp^1;
                l++;
                r--;
            }
        }
        return mat;
    }
}
